package es.us.dp1.lx_xy_24_25.Escape_From_Elba.map;

import org.springframework.messaging.simp.SimpMessagingTemplate;
import org.springframework.stereotype.Service;

import lombok.Data;

@Service
public class BattleService {

    private final SimpMessagingTemplate messagingTemplate;

    public BattleService(SimpMessagingTemplate messagingTemplate){
        this.messagingTemplate = messagingTemplate;
    }

    public BattleDTO startBattle(Integer matchId, Integer attackerId, Integer defenderId, Integer roomId){
        BattleDTO b = new BattleDTO();
        b.setMatchId(matchId);
        b.setAttackerId(attackerId);
        b.setDefenderId(defenderId);
        b.setRoomId(roomId);
        b.setStarted(true);
        b.setMessage("Battle started between " + attackerId + " and " + defenderId);
        messagingTemplate.convertAndSend(String.format("/topic/matches/%d/battles", matchId), b);
        return b;
    }

    @Data
    public static class BattleDTO{
        private Integer matchId;
        private Integer attackerId;
        private Integer defenderId;
        private Integer roomId;
        private boolean started;
        private String message;
    }
}
