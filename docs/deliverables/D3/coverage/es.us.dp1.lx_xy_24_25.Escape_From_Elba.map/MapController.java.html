package es.us.dp1.lx_xy_24_25.Escape_From_Elba.map;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import es.us.dp1.lx_xy_24_25.Escape_From_Elba.players.PlayerService;

@RestController
@RequestMapping("/api/v1/matches")
public class MapController {

    private final MapService mapService;
    private final PlayerService playerService;
    private final BattleService battleService;

    @Autowired
    public MapController(MapService mapService, PlayerService playerService, BattleService battleService){
        this.mapService = mapService;
        this.playerService = playerService;
        this.battleService = battleService;
    }

    @GetMapping("/{matchId}/map")
    public ResponseEntity<?> getMap(@PathVariable Integer matchId){
        MatchMapDTO dto = mapService.snapshot(matchId);
        if (dto==null) return ResponseEntity.notFound().build();
        return ResponseEntity.ok(dto);
    }

    @GetMapping("/{matchId}/players/{playerId}/position")
    public ResponseEntity<?> getPosition(@PathVariable Integer matchId, @PathVariable Integer playerId){
        PlayerPositionDTO p = mapService.findPlayerPosition(matchId, playerId);
        if (p==null) return ResponseEntity.notFound().build();
        return ResponseEntity.ok(p);
    }

    @PostMapping("/{matchId}/players/{playerId}/move")
    public ResponseEntity<?> move(@PathVariable Integer matchId, @PathVariable Integer playerId, @RequestBody MoveRequestDTO req){
        // Basic validation: ensure player exists and belongs to match
        // playerService can be extended to validate player in match; assume it has findById
        if (!playerService.findById(playerId).isPresent()){
            return ResponseEntity.badRequest().body(new MoveResponseDTO(false, null, false, null, "Player not found"));
        }

        MoveResponseDTO res = mapService.tryMove(matchId, playerId, req.getToRoomId());
        if (res.isBattleStarted()){
            // trigger battle
            PlayerPositionDTO defenderPos = mapService.findPlayerPosition(matchId, res.getOpponentPlayerId());
            Integer roomId = defenderPos!=null ? defenderPos.getRoomId() : req.getToRoomId();
            battleService.startBattle(matchId, playerId, res.getOpponentPlayerId(), roomId);
        }
        return ResponseEntity.ok(res);
    }

}
