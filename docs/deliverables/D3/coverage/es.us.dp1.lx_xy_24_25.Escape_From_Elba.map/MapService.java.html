package es.us.dp1.lx_xy_24_25.Escape_From_Elba.map;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.locks.ReentrantLock;

import org.springframework.messaging.simp.SimpMessagingTemplate;
import org.springframework.stereotype.Service;

import es.us.dp1.lx_xy_24_25.Escape_From_Elba.players.Player;

@Service
public class MapService {

    // matchId -> (roomId -> occupants)
    private final Map<Integer, Map<Integer, Set<Integer>>> occupants = new ConcurrentHashMap<>();
    private final Map<Integer, Map<Integer, List<Integer>>> adjacencies = new ConcurrentHashMap<>();
    private final Map<Integer, ReentrantLock> matchLocks = new ConcurrentHashMap<>();
    private final Map<Integer, Long> versions = new ConcurrentHashMap<>();

    private final SimpMessagingTemplate messagingTemplate;

    public MapService(SimpMessagingTemplate messagingTemplate) {
        this.messagingTemplate = messagingTemplate;
    }

    private ReentrantLock lockForMatch(Integer matchId){
        matchLocks.putIfAbsent(matchId, new ReentrantLock());
        return matchLocks.get(matchId);
    }

    public void initializeMap(Integer matchId, Map<Integer, List<Integer>> roomAdjacencies, Map<Integer, Integer> initialPositions){
        // set adjacencies
        adjacencies.put(matchId, new HashMap<>(roomAdjacencies));
        // create occupants map
        Map<Integer, Set<Integer>> roomOcc = new HashMap<>();
        for (Integer roomId : roomAdjacencies.keySet()){
            roomOcc.put(roomId, new HashSet<>());
        }
        occupants.put(matchId, roomOcc);
        // place players
        for (Map.Entry<Integer,Integer> e : initialPositions.entrySet()){
            Integer playerId = e.getKey();
            Integer roomId = e.getValue();
            roomOcc.get(roomId).add(playerId);
        }
        versions.put(matchId, System.currentTimeMillis());
        broadcastMap(matchId);
    }

    public MatchMapDTO snapshot(Integer matchId){
        Map<Integer, List<Integer>> adj = adjacencies.get(matchId);
        Map<Integer, Set<Integer>> occ = occupants.get(matchId);
        if (adj == null) return null;
        List<RoomDTO> rooms = new ArrayList<>();
        for (Integer roomId : adj.keySet()){
            List<Integer> adjacent = adj.get(roomId);
            Set<Integer> occSet = (occ!=null)? occ.get(roomId) : new HashSet<>();
            rooms.add(new RoomDTO(roomId, "room-"+roomId, adjacent, new ArrayList<>(occSet)));
        }
        return new MatchMapDTO(matchId, rooms, versions.getOrDefault(matchId, System.currentTimeMillis()));
    }

    public void broadcastMap(Integer matchId){
        MatchMapDTO snap = snapshot(matchId);
        if (snap != null){
            messagingTemplate.convertAndSend(String.format("/topic/matches/%d/map", matchId), snap);
        }
    }

    public PlayerPositionDTO findPlayerPosition(Integer matchId, Integer playerId){
        Map<Integer, Set<Integer>> occ = occupants.get(matchId);
        if (occ==null) return null;
        for (Map.Entry<Integer, Set<Integer>> e : occ.entrySet()){
            if (e.getValue().contains(playerId)){
                return new PlayerPositionDTO(playerId, e.getKey(), System.currentTimeMillis());
            }
        }
        return null;
    }

    // Attempt to move; returns MoveResponseDTO with battle if necessary
    public MoveResponseDTO tryMove(Integer matchId, Integer playerId, Integer toRoomId){
        ReentrantLock lock = lockForMatch(matchId);
        lock.lock();
        try{
            Map<Integer, List<Integer>> adj = adjacencies.get(matchId);
            if (adj==null) return new MoveResponseDTO(false, null, false, null, "Map not initialized");

            PlayerPositionDTO current = findPlayerPosition(matchId, playerId);
            if (current==null) return new MoveResponseDTO(false, null, false, null, "Player not in any room");

            Integer fromRoom = current.getRoomId();
            List<Integer> neighbors = adj.get(fromRoom);
            if (neighbors==null || !neighbors.contains(toRoomId)){
                return new MoveResponseDTO(false, null, false, null, "Destination not adjacent");
            }

            Map<Integer, Set<Integer>> rooms = occupants.get(matchId);
            Set<Integer> destOcc = rooms.get(toRoomId);
            if (destOcc==null) return new MoveResponseDTO(false, null, false, null, "Destination room unknown");

            if (destOcc.isEmpty()){
                // move
                rooms.get(fromRoom).remove(playerId);
                destOcc.add(playerId);
                versions.put(matchId, System.currentTimeMillis());
                broadcastMap(matchId);
                // broadcast position
                messagingTemplate.convertAndSend(String.format("/topic/matches/%d/positions", matchId), new PlayerPositionDTO(playerId, toRoomId, System.currentTimeMillis()));
                return new MoveResponseDTO(true, toRoomId, false, null, null);
            } else {
                // someone is here -> battle with first occupant
                Integer opponent = destOcc.iterator().next();
                return new MoveResponseDTO(false, null, true, opponent, "Battle started");
            }
        } finally {
            lock.unlock();
        }
    }
}
